rules_version = '2';
service cloud.firestore {
 match /databases/{database}/documents {
  function signedIn() {return request.auth != null;}
  function owner(uid) {return signedIn() && request.auth.uid == uid;}
  match /topics/{id} {allow read: if signedIn(); allow write: if false;}
  match /words/{id} {allow read: if signedIn(); allow write: if false;}
  match /users/{uid} {
   allow get, list: if owner(uid);
   allow create, update: if owner(uid) && request.resource.data.userId == uid && request.resource.data.dailyGoal is int && request.resource.data.dailyGoal >= 5 && request.resource.data.dailyGoal <= 20;
   allow delete: if false;
  }
  match /userProgress/{id} {
   allow read: if signedIn() && resource.data.userId == request.auth.uid;
   allow create, update: if signedIn() && request.resource.data.userId == request.auth.uid && id == request.auth.uid + '_' + request.resource.data.wordId && request.resource.data.level is int && request.resource.data.level >= 0 && request.resource.data.level <= 5;
   allow delete: if false;
  }
  match /reviewLogs/{id} {
   allow read: if signedIn() && resource.data.userId == request.auth.uid;
   allow create: if signedIn() && request.resource.data.userId == request.auth.uid && request.resource.data.rating in ['FORGOT','HARD','REMEMBER'];
   allow update, delete: if false;
  }
 }
}
